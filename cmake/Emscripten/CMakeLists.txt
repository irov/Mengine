CMAKE_MINIMUM_REQUIRED(VERSION 3.24)

SET(MENGINE_PROJECT_NAME "Mengine_Emscripten" CACHE STRING "Browser project name")

PROJECT(${MENGINE_PROJECT_NAME})

SET(MENGINE_REPOSITORY ${CMAKE_CURRENT_LIST_DIR}/../..)
SET(MENGINE_DEPENDENCIES_PROJECT Depends_EMSCRIPTEN)
SET(MENGINE_BUILD_MENGINE_SCRIPT_EMBEDDED ON)

INCLUDE(${MENGINE_REPOSITORY}/cmake/build_options_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/dev_options_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/base_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/macro_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/emscripten_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/mengine_template.cmake)

SET(CMAKE_COMPILE_WARNING_AS_ERROR ON)

SET_MENGINE_ENVIRONMENT(NATIVE EMSCRIPTEN OPENGL EMSCRIPTEN EMSCRIPTEN)

ADD_MENGINE_FRAMEWORK()

# platform
ADD_PLATFORM(EmscriptenPlatform "MENGINE_PLATFORM")

# systems
ADD_SYSTEM(MENGINE_SYSTEM_ALLOCATOR POSIXAllocatorSystem "MENGINE_SYSTEM_ALLOCATOR")
ADD_SYSTEM(MENGINE_SYSTEM_SOUND WebAudioSoundSystem "MENGINE_SYSTEM_SOUND")
ADD_SYSTEM(MENGINE_SYSTEM_FILE POSIXFileSystem "MENGINE_SYSTEM_FILE")
ADD_SYSTEM(MENGINE_SYSTEM_DATETIME POSIXDateTimeSystem "MENGINE_SYSTEM_DATETIME")
ADD_SYSTEM(MENGINE_SYSTEM_THREAD EmscriptenThreadSystem "MENGINE_SYSTEM_THREAD")
ADD_SYSTEM(MENGINE_SYSTEM_TIME POSIXTimeSystem "MENGINE_SYSTEM_TIME")
ADD_SYSTEM(MENGINE_SYSTEM_CRYPTOGRAPHY POSIXCryptographySystem "MENGINE_SYSTEM_CRYPTOGRAPHY")
ADD_SYSTEM(MENGINE_SYSTEM_LANGUAGE POSIXLanguageSystem "MENGINE_SYSTEM_LANGUAGE")
ADD_SYSTEM(MENGINE_SYSTEM_SOCKET OFF "MENGINE_SYSTEM_SOCKET")
ADD_SYSTEM(MENGINE_SYSTEM_HTTP OFF "MENGINE_SYSTEM_HTTP")
ADD_SYSTEM(MENGINE_SYSTEM_RENDER OpenGLRenderSystem "MENGINE_SYSTEM_RENDER")

# plugins
ADD_PLUGIN(MENGINE_PLUGIN_MENGINE ON OFF "MENGINE_PLUGIN_MENGINE")
ADD_PLUGIN(MENGINE_PLUGIN_PYTHONFRAMEWORK ON OFF "MENGINE_PLUGIN_PYTHONFRAMEWORK")
ADD_PLUGIN(MENGINE_PLUGIN_RESOURCEPREFETCHER ON OFF "MENGINE_PLUGIN_RESOURCEPREFETCHER")
ADD_PLUGIN(MENGINE_PLUGIN_METABUF ON OFF "MENGINE_PLUGIN_METABUF")
ADD_PLUGIN(MENGINE_PLUGIN_IMAGE_CODEC ON OFF "MENGINE_PLUGIN_IMAGE_CODEC")
ADD_PLUGIN(MENGINE_PLUGIN_WAV ON OFF "MENGINE_PLUGIN_WAV")
ADD_PLUGIN(MENGINE_PLUGIN_AMPLIFIER ON OFF "MENGINE_PLUGIN_AMPLIFIER")
ADD_PLUGIN(MENGINE_PLUGIN_OGG_VORBIS ON OFF "MENGINE_PLUGIN_OGG_VORBIS")
ADD_PLUGIN(MENGINE_PLUGIN_MP3 ON OFF "MENGINE_PLUGIN_MP3")
ADD_PLUGIN(MENGINE_PLUGIN_ZIP ON OFF "MENGINE_PLUGIN_ZIP")
ADD_PLUGIN(MENGINE_PLUGIN_LZ4 ON OFF "MENGINE_PLUGIN_LZ4")
ADD_PLUGIN(MENGINE_PLUGIN_JSON ON OFF "MENGINE_PLUGIN_JSON")
ADD_PLUGIN(MENGINE_PLUGIN_JPEG ON OFF "MENGINE_PLUGIN_JPEG")
ADD_PLUGIN(MENGINE_PLUGIN_PNG ON OFF "MENGINE_PLUGIN_PNG")
ADD_PLUGIN(MENGINE_PLUGIN_MOVIE ON OFF "MENGINE_PLUGIN_MOVIE")
ADD_PLUGIN(MENGINE_PLUGIN_VIDEO ON OFF "MENGINE_PLUGIN_VIDEO")
ADD_PLUGIN(MENGINE_PLUGIN_THEORA OFF OFF "MENGINE_PLUGIN_THEORA")
ADD_PLUGIN(MENGINE_PLUGIN_BITMAPFONT ON OFF "MENGINE_PLUGIN_BITMAPFONT")
ADD_PLUGIN(MENGINE_PLUGIN_TTF ON OFF "MENGINE_PLUGIN_TTF")
ADD_PLUGIN(MENGINE_PLUGIN_FE ON OFF "MENGINE_PLUGIN_FE")
ADD_PLUGIN(MENGINE_PLUGIN_FONTEFFECT ON OFF "MENGINE_PLUGIN_FONTEFFECT")
ADD_PLUGIN_IF(MENGINE_PLUGIN_ASTRALAX OFF OFF "MENGINE_PLUGIN_ASTRALAX" "${THIRDPARTY_DIR}/astralax_emscripten")

MENGINE_ADD_DEFINITION(MENGINE_RENDER_TEXTURE_RGBA)
MENGINE_ADD_DEFINITION(MENGINE_RENDER_COLOR_RGBA)
MENGINE_ADD_DEFINITION(MENGINE_OPENGL_RENDER_CHECK_ERROR 0)

SET_MENGINE_SOLUTIONS_OUTPUT_DIRECTORY()

ADD_SUBDIRECTORY(${MENGINE_SOURCE_DIR} ${CMAKE_TEMP_DIR}/src)

ADD_SUBDIRECTORY(${MENGINE_SOURCE_DIR}/Applications/EmscriptenApplication ${CMAKE_TEMP_DIR}/Applications/EmscriptenApplication)
