CMAKE_MINIMUM_REQUIRED(VERSION 3.20)

SET(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "Apple Silicon tools architecture" FORCE)
SET(CMAKE_SYSTEM_PROCESSOR "arm64" CACHE STRING "Apple Silicon processor" FORCE)
SET(MENGINE_PROJECT_NAME "Mengine_Tools_Xcode_MacOS" CACHE STRING "Mengine project name" FORCE)

PROJECT(${MENGINE_PROJECT_NAME})

SET(MENGINE_REPOSITORY ${CMAKE_CURRENT_LIST_DIR}/../..)

INCLUDE(${MENGINE_REPOSITORY}/cmake/build_options_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/dev_options_template.cmake)

SET(MENGINE_BUILD_MENGINE_SCRIPT_EMBEDDED OFF CACHE BOOL "MENGINE_BUILD_MENGINE_SCRIPT_EMBEDDED" FORCE)

INCLUDE(${MENGINE_REPOSITORY}/cmake/base_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/macro_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/xcode_macos_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/mengine_template.cmake)

# The supplied Astralax runtime is built with macOS 15.0 as its minimum.
SET(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Tools deployment target" FORCE)

SET_MENGINE_ENVIRONMENT(MACOS METAL MACOS XCODE)

if(NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "Mengine tools for macOS support arm64 only")
endif()

SET(CMAKE_TEMP_DIR ${CMAKE_BINARY_DIR}/Tools)

# Tool plugins are linked statically so the installed commands do not depend on
# a separately deployed Mengine plugin directory.
ADD_PLUGIN(MENGINE_PLUGIN_DEVELOPMENTCONVERTER ON OFF "MENGINE_PLUGIN_DEVELOPMENTCONVERTER")
ADD_PLUGIN(MENGINE_PLUGIN_METABUF ON OFF "MENGINE_PLUGIN_METABUF")
ADD_PLUGIN(MENGINE_PLUGIN_IMAGE_CODEC ON OFF "MENGINE_PLUGIN_IMAGE_CODEC")
ADD_PLUGIN(MENGINE_PLUGIN_ZIP ON OFF "MENGINE_PLUGIN_ZIP")
ADD_PLUGIN(MENGINE_PLUGIN_LZ4 ON OFF "MENGINE_PLUGIN_LZ4")
ADD_PLUGIN(MENGINE_PLUGIN_MOVIE ON OFF "MENGINE_PLUGIN_MOVIE")
ADD_PLUGIN(MENGINE_PLUGIN_JPEG ON OFF "MENGINE_PLUGIN_JPEG")
ADD_PLUGIN(MENGINE_PLUGIN_PNG ON OFF "MENGINE_PLUGIN_PNG")
ADD_PLUGIN(MENGINE_PLUGIN_PVRTC ON OFF "MENGINE_PLUGIN_PVRTC")
ADD_PLUGIN(MENGINE_PLUGIN_WEBP ON OFF "MENGINE_PLUGIN_WEBP")
ADD_PLUGIN(MENGINE_PLUGIN_THEORA ON OFF "MENGINE_PLUGIN_THEORA")
ADD_PLUGIN(MENGINE_PLUGIN_OGG_VORBIS ON OFF "MENGINE_PLUGIN_OGG_VORBIS")
ADD_PLUGIN(MENGINE_PLUGIN_DAZZLE ON OFF "MENGINE_PLUGIN_DAZZLE")
ADD_PLUGIN(MENGINE_PLUGIN_JSON OFF OFF "MENGINE_PLUGIN_JSON")

SET_MENGINE_SOLUTIONS_OUTPUT_DIRECTORY()

INCLUDE_DIRECTORIES(${MENGINE_SOURCE_DIR})
INCLUDE_DIRECTORIES(${MENGINE_SOURCE_DIR}/Tools)
INCLUDE_DIRECTORIES(${THIRDPARTY_DIR}/libmath/include)
INCLUDE_DIRECTORIES(${THIRDPARTY_DIR}/stdex/include)

ADD_SUBDIRECTORY(${MENGINE_SOURCE_DIR}/Environment ${CMAKE_TEMP_DIR}/Environment)
ADD_SUBDIRECTORY(${MENGINE_SOURCE_DIR}/PrecompiledHeader ${CMAKE_TEMP_DIR}/PrecompiledHeader)
ADD_SUBDIRECTORY(${MENGINE_SOURCE_DIR}/Config ${CMAKE_TEMP_DIR}/Config)
ADD_SUBDIRECTORY(${MENGINE_SOURCE_DIR}/Kernel ${CMAKE_TEMP_DIR}/Kernel)
ADD_SUBDIRECTORY(${MENGINE_SOURCE_DIR}/Engine ${CMAKE_TEMP_DIR}/Engine)
ADD_SUBDIRECTORY(${MENGINE_SOURCE_DIR}/Metacode ${CMAKE_TEMP_DIR}/Metacode)

set(SOURCE_DIR_SYSTEMS ${MENGINE_SOURCE_DIR}/Systems)
set(TEMP_DIR_SYSTEMS ${CMAKE_TEMP_DIR}/Systems)
set(SOURCE_DIR_SERVICES ${MENGINE_SOURCE_DIR}/Services)
set(TEMP_DIR_SERVICES ${CMAKE_TEMP_DIR}/Services)
set(SOURCE_DIR_PLUGINS ${MENGINE_SOURCE_DIR}/Plugins)
set(TEMP_DIR_PLUGINS ${CMAKE_TEMP_DIR}/Plugins)

ADD_SUBDIRECTORY(${SOURCE_DIR_SYSTEMS}/POSIXTimeSystem ${TEMP_DIR_SYSTEMS}/POSIXTimeSystem)
ADD_SUBDIRECTORY(${SOURCE_DIR_SYSTEMS}/POSIXThreadSystem ${TEMP_DIR_SYSTEMS}/POSIXThreadSystem)
ADD_SUBDIRECTORY(${SOURCE_DIR_SYSTEMS}/POSIXAllocatorSystem ${TEMP_DIR_SYSTEMS}/POSIXAllocatorSystem)
ADD_SUBDIRECTORY(${SOURCE_DIR_SYSTEMS}/POSIXDateTimeSystem ${TEMP_DIR_SYSTEMS}/POSIXDateTimeSystem)
ADD_SUBDIRECTORY(${SOURCE_DIR_SYSTEMS}/AppleFileSystem ${TEMP_DIR_SYSTEMS}/AppleFileSystem)

foreach(SERVICE
    DocumentService
    NotificationService
    EnumeratorService
    FactoryService
    ProviderService
    StringizeService
    ArchiveService
    FileService
    CodecService
    DataService
    MemoryService
    ConverterService
    OptionsService
    LoggerService
    PluginService
    ThreadService
    ConfigService
    TextService
    PrototypeService
    VocabularyService
    TimelineService
    TimepipeService
    ModuleService
)
    ADD_SUBDIRECTORY(${SOURCE_DIR_SERVICES}/${SERVICE} ${TEMP_DIR_SERVICES}/${SERVICE})
endforeach()

ADD_SUBDIRECTORY(${MENGINE_SOURCE_DIR}/Platforms/MacOSPlatform ${CMAKE_TEMP_DIR}/Platforms/MacOSPlatform)

foreach(PLUGIN
    DevelopmentConverterPlugin
    ImageCodecPlugin
    ZipPlugin
    LZ4Plugin
    MoviePlugin
    JPEGPlugin
    PNGPlugin
    PVRTCPlugin
    WebPPlugin
    TheoraPlugin
    OggVorbisPlugin
    MetabufPlugin
    DazzlePlugin
)
    ADD_SUBDIRECTORY(${SOURCE_DIR_PLUGINS}/${PLUGIN} ${TEMP_DIR_PLUGINS}/${PLUGIN})
endforeach()

ADD_SUBDIRECTORY(${MENGINE_SOURCE_DIR}/Tools ${CMAKE_TEMP_DIR}/Tools)
