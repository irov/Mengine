CMAKE_MINIMUM_REQUIRED(VERSION 3.25)

SET(MENGINE_PROJECT_NAME "Mengine_Xcode_MacOS_WebView" CACHE STRING "Mengine project name")

PROJECT(${MENGINE_PROJECT_NAME} C CXX)

SET(MENGINE_REPOSITORY ${CMAKE_CURRENT_LIST_DIR}/../..)

INCLUDE(${MENGINE_REPOSITORY}/cmake/build_options_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/master_options_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/base_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/macro_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/xcode_macos_template.cmake)
INCLUDE(${MENGINE_REPOSITORY}/cmake/mengine_template.cmake)

SET(CMAKE_COMPILE_WARNING_AS_ERROR ON)

SET_MENGINE_ENVIRONMENT(WEBVIEW MACOS NONE MACOS XCODE)

ADD_PLATFORM(MacOSWebViewPlatform "MENGINE_PLATFORM")

ADD_SYSTEM(MENGINE_SYSTEM_ALLOCATOR POSIXAllocatorSystem "MENGINE_SYSTEM_ALLOCATOR")
ADD_SYSTEM(MENGINE_SYSTEM_FILE AppleFileSystem "MENGINE_SYSTEM_FILE")
ADD_SYSTEM(MENGINE_SYSTEM_DATETIME POSIXDateTimeSystem "MENGINE_SYSTEM_DATETIME")
ADD_SYSTEM(MENGINE_SYSTEM_THREAD POSIXThreadSystem "MENGINE_SYSTEM_THREAD")
ADD_SYSTEM(MENGINE_SYSTEM_TIME POSIXTimeSystem "MENGINE_SYSTEM_TIME")
ADD_SYSTEM(MENGINE_SYSTEM_CRYPTOGRAPHY AppleCryptographySystem "MENGINE_SYSTEM_CRYPTOGRAPHY")
ADD_SYSTEM(MENGINE_SYSTEM_HTTP AppleHttpSystem "MENGINE_SYSTEM_HTTP")
ADD_SYSTEM(MENGINE_SYSTEM_SOCKET POSIXSocketSystem "MENGINE_SYSTEM_SOCKET")

SET_MENGINE_SOLUTIONS_OUTPUT_DIRECTORY()

ADD_SUBDIRECTORY(${MENGINE_SOURCE_DIR} ${CMAKE_TEMP_DIR}/src)

ADD_SUBDIRECTORY(${MENGINE_SOURCE_DIR}/Applications/MacOSWebViewApplication ${CMAKE_TEMP_DIR}/Applications/MacOSWebViewApplication)
