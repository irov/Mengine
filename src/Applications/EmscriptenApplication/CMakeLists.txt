MENGINE_PROJECT(EmscriptenApplication)

ADD_FILTER(
src
    EmscriptenApplication.cpp
    EmscriptenApplication.h
    EmscriptenApplicationExport.cpp
    EmscriptenMain.cpp
)

ADD_MENGINE_EXECUTABLE_UNIX()

SET_TARGET_PROPERTIES(${PROJECT_NAME} PROPERTIES
    OUTPUT_NAME Mengine
    SUFFIX ".html"
)

TARGET_LINK_LIBRARIES(${PROJECT_NAME}
    Kernel
    MengineNativeRuntime
    ${APPLICATION_LINK_LIBRARIES}
)

IF(APPLICATION_DEPENDENCIES)
    ADD_DEPENDENCIES(${PROJECT_NAME} ${APPLICATION_DEPENDENCIES})
ENDIF()

TARGET_LINK_OPTIONS(${PROJECT_NAME} PRIVATE
    -sENVIRONMENT=web
    -sEXIT_RUNTIME=0
    "-sALLOW_MEMORY_GROWTH=$<BOOL:${MENGINE_EMSCRIPTEN_ALLOW_MEMORY_GROWTH}>"
    "-sSTACK_SIZE=${MENGINE_EMSCRIPTEN_STACK_SIZE}"
    "-sINITIAL_MEMORY=${MENGINE_EMSCRIPTEN_INITIAL_MEMORY}"
    "-sMIN_WEBGL_VERSION=${MENGINE_EMSCRIPTEN_MIN_WEBGL_VERSION}"
    "-sMAX_WEBGL_VERSION=${MENGINE_EMSCRIPTEN_MAX_WEBGL_VERSION}"
    "-sGL_ENABLE_GET_PROC_ADDRESS=$<BOOL:${MENGINE_EMSCRIPTEN_GL_ENABLE_GET_PROC_ADDRESS}>"
    -sFORCE_FILESYSTEM=1
    -lidbfs.js
    "SHELL:--pre-js '${CMAKE_CURRENT_SOURCE_DIR}/EmscriptenPersistentUserStorage.js'"
    "$<$<CONFIG:Debug>:-sASSERTIONS=1>"
)

IF(MENGINE_EMSCRIPTEN_ALLOW_MEMORY_GROWTH)
    TARGET_LINK_OPTIONS(${PROJECT_NAME} PRIVATE
        "-sMAXIMUM_MEMORY=${MENGINE_EMSCRIPTEN_MAXIMUM_MEMORY}"
    )
ENDIF()

IF(MENGINE_EMSCRIPTEN_MAX_WEBGL_VERSION GREATER_EQUAL 2)
    TARGET_LINK_OPTIONS(${PROJECT_NAME} PRIVATE
        -sWEBGL2_BACKWARDS_COMPATIBILITY_EMULATION=1
    )
ENDIF()

SET_PROPERTY(TARGET ${PROJECT_NAME} APPEND PROPERTY LINK_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/EmscriptenPersistentUserStorage.js
)

IF(MENGINE_DEPLOY_PATH)
    IF(EXISTS "${MENGINE_DEPLOY_PATH}/project_deploy.cmake")
        INCLUDE("${MENGINE_DEPLOY_PATH}/project_deploy.cmake")
    ENDIF()
ENDIF()
