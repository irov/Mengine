MENGINE_PROJECT(EmscriptenPlatform)

ADD_FILTER(
src
    EmscriptenPlatformBridge.h
    EmscriptenPlatformService.h
    EmscriptenPlatformService.cpp
    EmscriptenPlatformSystem.h
    EmscriptenPlatformSystem.cpp
)

ADD_MENGINE_LIBRARY(Platforms)

TARGET_COMPILE_DEFINITIONS(${PROJECT_NAME} PRIVATE
    MENGINE_EMSCRIPTEN_MIN_WEBGL_VERSION=${MENGINE_EMSCRIPTEN_MIN_WEBGL_VERSION}
    MENGINE_EMSCRIPTEN_MAX_WEBGL_VERSION=${MENGINE_EMSCRIPTEN_MAX_WEBGL_VERSION}
)

TARGET_LINK_LIBRARIES(${PROJECT_NAME} Kernel)

TARGET_LINK_OPTIONS(${PROJECT_NAME} INTERFACE
    "SHELL:--js-library '${CMAKE_CURRENT_SOURCE_DIR}/EmscriptenPlatformBridge.js'"
)

SET_PROPERTY(TARGET ${PROJECT_NAME} APPEND PROPERTY INTERFACE_LINK_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/EmscriptenPlatformBridge.js
)
