MENGINE_PROJECT(iOSOpButtonPlugin)

ADD_FILTER(
src
    iOSOpButtonPlugin.h
    iOSOpButtonPlugin.mm
)

if(MENGINE_BUILD_MENGINE_SCRIPT_EMBEDDED)
    ADD_FILTER(
    embedding
        iOSOpButtonScriptEmbedding.h
        iOSOpButtonScriptEmbedding.mm
    )

    INCLUDE_DIRECTORIES(${THIRDPARTY_DIR}/pybind/include)
endif()

ADD_MENGINE_PLUGIN(MENGINE_PLUGIN_IOS_OPBUTTON)
ADD_MENGINE_APPLE_APPLICATION_DELEGATE("iOSOpButtonPlugin")

SET(MENGINE_IOS_OPBUTTON_SDK_GIT "https://github.com/opbutton-org/opbutton-sdk-swift.git" CACHE STRING "OpButton SDK repository")
SET(MENGINE_IOS_OPBUTTON_SDK_VERSION "1.0.66" CACHE STRING "OpButton SDK version")
ADD_MENGINE_SWIFT_PACKAGE("OpButtonSDK" "${MENGINE_IOS_OPBUTTON_SDK_GIT}" "${MENGINE_IOS_OPBUTTON_SDK_VERSION}")

TARGET_COMPILE_OPTIONS(${PROJECT_NAME} PRIVATE -fcxx-modules)
TARGET_LINK_LIBRARIES(${PROJECT_NAME} Kernel)

if(MENGINE_BUILD_MENGINE_SCRIPT_EMBEDDED)
    TARGET_LINK_LIBRARIES(${PROJECT_NAME} PythonEnvironment)
endif()

if(MENGINE_PLUGIN_IOS_USERNOTIFICATIONCENTER)
    TARGET_LINK_LIBRARIES(${PROJECT_NAME} iOSUserNotificationCenterPlugin)
endif()
